package com.conforttravel.api.aerolinea.controller;

import com.conforttravel.api.aerolinea.dto.AerolineaFilterRequest;
import com.conforttravel.api.aerolinea.dto.AerolineaRequest;
import com.conforttravel.api.aerolinea.dto.AerolineaResponse;
import com.conforttravel.api.aerolinea.service.AerolineaService;
import com.conforttravel.api.shared.dto.PageResponse;
import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import org.springframework.data.domain.Pageable;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@RequestMapping("/api/aerolineas")
@RequiredArgsConstructor
public class AerolineaController {

    private final AerolineaService aerolineaService;

    @GetMapping
    public ResponseEntity<List<AerolineaResponse>> obtenerTodos() {
        return ResponseEntity.ok(aerolineaService.obtenerTodos());
    }

    @GetMapping("/{id}")
    public ResponseEntity<AerolineaResponse> obtenerPorId(
            @PathVariable Long id
    ) {
        return ResponseEntity.ok(
                aerolineaService.obtenerPorId(id)
        );
    }

    @PostMapping
    public ResponseEntity<AerolineaResponse> crear(
            @Valid @RequestBody AerolineaRequest request
    ) {
        return ResponseEntity
                .status(HttpStatus.CREATED)
                .body(aerolineaService.crear(request));
    }

    @PutMapping("/{id}")

            FilterChain filterChain
    ) throws ServletException, IOException {

        final String authorizationHeader =
                request.getHeader("Authorization");

        if (authorizationHeader == null
                || !authorizationHeader.startsWith("Bearer ")) {

            filterChain.doFilter(request, response);
            return;
        }

        final String jwt =
                authorizationHeader.substring(7);

        try {

            final String username =
                    jwtService.extractUsername(jwt);

            if (username != null
                    && SecurityContextHolder
                    .getContext()
                    .getAuthentication() == null) {

                UserDetails userDetails =
                        userDetailsService.loadUserByUsername(username);

                if (jwtService.isTokenValid(jwt, userDetails)) {

                    UsernamePasswordAuthenticationToken authentication =
                            new UsernamePasswordAuthenticationToken(
                                    userDetails,
                                    null,
                                    userDetails.getAuthorities()
                            );

                    authentication.setDetails(
                            new WebAuthenticationDetailsSource()
                                    .buildDetails(request)
                    );

                    SecurityContextHolder
                            .getContext()
                            .setAuthentication(authentication);
                }
            }

        } catch (Exception ignored) {
            /*
             * Un token inválido no debe detener el filtro.
             * Spring Security determinará posteriormente
             * que la solicitud no está autenticada.
             */
        }

        filterChain.doFilter(request, response);
    }
}

package com.conforttravel.api.auth.security;

import io.jsonwebtoken.Claims;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.stereotype.Service;

import javax.crypto.SecretKey;
import java.nio.charset.StandardCharsets;
import java.util.Date;
import java.util.HashMap;
import java.util.Map;
import java.util.function.Function;

@Service
public class JwtService {

    private final String secret;
    private final long expiration;

    public JwtService(
            @Value("${security.jwt.secret}") String secret,
            @Value("${security.jwt.expiration}") long expiration
    ) {
        this.secret = secret;
        this.expiration = expiration;
    }

    public String generateToken(UserDetails userDetails) {

        Map<String, Object> claims = new HashMap<>();

        claims.put(
                "roles",
                userDetails.getAuthorities()
                        .stream()
                        .map(authority -> authority.getAuthority())
                        .toList()
        );

        return Jwts.builder()
                .claims(claims)
                .subject(userDetails.getUsername())
                .issuedAt(new Date())
                .expiration(new Date(System.currentTimeMillis() + expiration))
                .signWith(getSigningKey())
                .compact();
    }

    public String extractUsername(String token) {
        return extractClaim(token, Claims::getSubject);
    }

    public Date extractExpiration(String token) {
        return extractClaim(token, Claims::getExpiration);
    }

    public <T> T extractClaim(
            String token,
            Function<Claims, T> claimsResolver
    ) {
        Claims claims = extractAllClaims(token);
        return claimsResolver.apply(claims);
    }

    public boolean isTokenValid(
            String token,
            UserDetails userDetails
    ) {
        final String username = extractUsername(token);

        return username.equals(userDetails.getUsername())
                && !isTokenExpired(token);
    }

    private boolean isTokenExpired(String token) {
        return extractExpiration(token).before(new Date());
    }

    private Claims extractAllClaims(String token) {

        return Jwts.parser()
                .verifyWith(getSigningKey())
                .build()
                .parseSignedClaims(token)
                .getPayload();
    }

    private SecretKey getSigningKey() {

        byte[] keyBytes = secret.getBytes(StandardCharsets.UTF_8);

        return Keys.hmacShaKeyFor(keyBytes);
    }
}

    @Transactional

        List<CheckIn> checkInsDisponibles =
                checkInRepository
                        .findByEstadoAndFechaDisponibleLessThanEqual(
                                EstadoCheckIn.PENDIENTE,
                                ahora
                        );

            log.debug(
                    "Scheduler de check-in: no existen check-ins pendientes disponibles"
            );
            return;
        }

        Usuario usuario = obtenerUsuarioNotificacion();

        for (CheckIn checkIn : checkInsDisponibles) {

            if (checkIn.getEstado() != EstadoCheckIn.PENDIENTE) {
                continue;
            }

            crearNotificacion(checkIn, usuario);

            checkIn.setEstado(EstadoCheckIn.NOTIFICADO);
            checkIn.setFechaNotificacion(ahora);

            checkInRepository.save(checkIn);

            log.info(
                    "Check-in {} notificado automáticamente. Tramo: {}",
                    checkIn.getId(),
                    checkIn.getTramo().getId()
            );
        }
    }

    private void crearNotificacion(
            CheckIn checkIn,
            Usuario usuario
    ) {

        Notificacion notificacion = Notificacion.builder()
                .tipo(TipoNotificacion.CHECK_IN_DISPONIBLE)
                .mensaje(
                        "El check-in del tramo "
                                + checkIn.getTramo().getId()
                                + " ya se encuentra disponible."
                )
                .fechaCreacion(LocalDateTime.now())
                .usuario(usuario)
                .checkIn(checkIn)
                .build();

        notificacionRepository.save(notificacion);
    }

        );

import com.conforttravel.api.servicio.repository.ServicioAdicionalRepository;
import com.conforttravel.api.servicio.service.ServicioAdicionalService;
import com.conforttravel.api.shared.exception.ResourceNotFoundException;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true)
public class ServicioAdicionalServiceImpl
        implements ServicioAdicionalService {

    private final ServicioAdicionalRepository servicioRepository;
    private final ServicioAdicionalMapper servicioMapper;

    @Override
    @Transactional
    public ServicioAdicionalResponse crear(
            ServicioAdicionalRequest request
    ) {
        ServicioAdicional servicio = servicioMapper.toEntity(request);

        if (servicio.getActivo() == null) {
            servicio.setActivo(true);
        }

        ServicioAdicional guardado = servicioRepository.save(servicio);

        return servicioMapper.toResponseDto(guardado);
    }

    @Override
    public ServicioAdicionalResponse obtenerPorId(Long id) {
        return servicioMapper.toResponseDto(obtenerEntidad(id));
    }

    @Override
    public List<ServicioAdicionalResponse> obtenerTodos() {
        return servicioRepository.findAll()
                .stream()
                .map(servicioMapper::toResponseDto)
                .toList();
    }

    @Override
    @Transactional
    public ServicioAdicionalResponse actualizar(
            Long id,
            ServicioAdicionalRequest request
    ) {
        ServicioAdicional servicio = obtenerEntidad(id);

        servicio.setNombre(request.nombre());
        servicio.setDescripcion(request.descripcion());

        if (request.activo() != null) {
            servicio.setActivo(request.activo());
        }

        ServicioAdicional actualizado = servicioRepository.save(servicio);

        return servicioMapper.toResponseDto(actualizado);
    }

    @Override
    @Transactional
    public void eliminar(Long id) {
        ServicioAdicional servicio = obtenerEntidad(id);

        servicioRepository.delete(servicio);
    }

    private ServicioAdicional obtenerEntidad(Long id) {
        return servicioRepository.findById(id)
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "Servicio adicional no encontrado con id: " + id
                        )
                );
    }
}

                HttpStatus.NOT_FOUND,

    @ExceptionHandler(BusinessException.class)
    public ResponseEntity<ApiErrorResponse> handleBusinessException(

        return buildResponse(

                "BUSINESS_RULE_ERROR",
                exception.getMessage(),
                request.getRequestURI()
        );
    }

    @ExceptionHandler(ConflictException.class)
    public ResponseEntity<ApiErrorResponse> handleConflict(
            ConflictException exception,
            HttpServletRequest request
    ) {
        return buildResponse(
                HttpStatus.CONFLICT,
                "CONFLICT",
                exception.getMessage(),
                request.getRequestURI()
        );
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ApiErrorResponse> handleValidation(
            MethodArgumentNotValidException exception,
            HttpServletRequest request
    ) {
        String message = exception.getBindingResult()
                .getFieldErrors()
                .stream()
                .map(error -> error.getField() + ": " + error.getDefaultMessage())
                .collect(Collectors.joining("; "));

        return buildResponse(
                HttpStatus.BAD_REQUEST,
                "VALIDATION_ERROR",
                message,
                request.getRequestURI()
        );
    }

    @ExceptionHandler(ConstraintViolationException.class)
    public ResponseEntity<ApiErrorResponse> handleConstraintViolation(
            ConstraintViolationException exception,
            HttpServletRequest request
    ) {
        String message = exception.getConstraintViolations()
                .stream()
                .map(violation ->
                        violation.getPropertyPath() + ": " + violation.getMessage()
                )
                .collect(Collectors.joining("; "));

        return buildResponse(
                HttpStatus.BAD_REQUEST,
                "VALIDATION_ERROR",
                message,
                request.getRequestURI()
        );
    }

package com.conforttravel.api.tramo.service.impl;

import com.conforttravel.api.shared.exception.BusinessException;
import com.conforttravel.api.shared.exception.ResourceNotFoundException;
import com.conforttravel.api.tramo.dto.TramoRequest;
import com.conforttravel.api.tramo.dto.TramoResponse;
import com.conforttravel.api.tramo.entity.Tramo;
import com.conforttravel.api.tramo.mapper.TramoMapper;
import com.conforttravel.api.tramo.repository.TramoRepository;
import com.conforttravel.api.tramo.service.TramoService;
import com.conforttravel.api.viaje.entity.Viaje;
import com.conforttravel.api.viaje.repository.ViajeRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true)
public class TramoServiceImpl implements TramoService {

    private final TramoRepository tramoRepository;
    private final ViajeRepository viajeRepository;
    private final TramoMapper tramoMapper;

    @Override
    @Transactional
    public TramoResponse crear(TramoRequest request) {
        validarFechas(request);

        Viaje viaje = obtenerViaje(request.viajeId());

        Tramo tramo = tramoMapper.toEntity(request);
        tramo.setViaje(viaje);

        Tramo guardado = tramoRepository.save(tramo);

        return tramoMapper.toResponseDto(guardado);
    }

    @Override
    public TramoResponse obtenerPorId(Long id) {
        return tramoMapper.toResponseDto(obtenerEntidad(id));
    }

    @Override
    public List<TramoResponse> obtenerTodos() {
        return tramoRepository.findAll()
                .stream()
                .map(tramoMapper::toResponseDto)
                .toList();
    }

    @Override
    @Transactional
    public TramoResponse actualizar(
            Long id,
            TramoRequest request
    ) {
        validarFechas(request);

        Tramo tramo = obtenerEntidad(id);
        Viaje viaje = obtenerViaje(request.viajeId());

        tramo.setTipoTramo(request.tipoTramo());
        tramo.setTipoViaje(request.tipoViaje());
        tramo.setOrigen(request.origen());
        tramo.setDestino(request.destino());
        tramo.setFechaHoraSalida(request.fechaHoraSalida());
        tramo.setFechaHoraLlegada(request.fechaHoraLlegada());
        tramo.setViaje(viaje);

        Tramo actualizado = tramoRepository.save(tramo);

        return tramoMapper.toResponseDto(actualizado);
    }

    @Override
    @Transactional
    public void eliminar(Long id) {
        Tramo tramo = obtenerEntidad(id);

        tramoRepository.delete(tramo);
    }

    private Tramo obtenerEntidad(Long id) {
        return tramoRepository.findById(id)
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "Tramo no encontrado con id: " + id
                        )
                );
    }

    private Viaje obtenerViaje(Long id) {
        return viajeRepository.findById(id)
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "Viaje no encontrado con id: " + id
                        )
                );
    }

    private void validarFechas(TramoRequest request) {
        if (request.fechaHoraLlegada() != null
                && !request.fechaHoraLlegada()
                .isAfter(request.fechaHoraSalida())) {

            throw new BusinessException(
                    "La fecha/hora de llegada debe ser posterior "
                            + "a la fecha/hora de salida"
            );
        }
    }
}

@Transactional(readOnly = true)

    private final UsuarioMapper usuarioMapper;
    private final PasswordEncoder passwordEncoder;

    @Override
    @Transactional
    public UsuarioResponse crear(UsuarioRequest request) {
        Usuario usuario = usuarioMapper.toEntity(request);

        /*
         * La gestión de passwordHash se completará junto con
         * la configuración de seguridad/autenticación.
         */
        usuario.setPasswordHash(passwordEncoder.encode(request.password()));

        if (usuario.getActivo() == null) {
            usuario.setActivo(true);
        }

        Usuario guardado = usuarioRepository.save(usuario);

        return usuarioMapper.toResponseDto(guardado);
    }

    @Override
    public UsuarioResponse obtenerPorId(Long id) {
        return usuarioMapper.toResponseDto(obtenerEntidad(id));
    }

    @Override
    public List<UsuarioResponse> obtenerTodos() {
        return usuarioRepository.findAll()
                .stream()
                .map(usuarioMapper::toResponseDto)
                .toList();
    }

    @Override
    @Transactional
    public UsuarioResponse actualizar(
            Long id,
            UsuarioRequest request
    ) {
        Usuario usuario = obtenerEntidad(id);

        usuario.setNombre(request.nombre());
        usuario.setApellido(request.apellido());
        usuario.setEmail(request.email());
        usuario.setRol(request.rol());

        if (request.activo() != null) {
            usuario.setActivo(request.activo());
        }

        usuario.setPasswordHash(
                passwordEncoder.encode(request.password())
        );

        Usuario actualizado = usuarioRepository.save(usuario);

        return usuarioMapper.toResponseDto(actualizado);
    }

    @Override
    @Transactional
    public void eliminar(Long id) {
        Usuario usuario = obtenerEntidad(id);

        usuarioRepository.delete(usuario);
    }

    private Usuario obtenerEntidad(Long id) {
        return usuarioRepository.findById(id)
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "Usuario no encontrado con id: " + id
                        )
                );
    }

package com.conforttravel.api.viaje.entity;

import com.conforttravel.api.tramo.entity.Tramo;
import com.conforttravel.api.venta.entity.Venta;
import jakarta.persistence.*;
import lombok.*;

import java.util.ArrayList;
import java.util.List;

@Entity
@Table(
        name = "viaje",
        uniqueConstraints = {
                @UniqueConstraint(
                        name = "uk_viaje_venta",
                        columnNames = "venta_id"
                )
        }
)
@Getter
@Setter
@NoArgsConstructor
@AllArgsConstructor
@Builder
public class Viaje {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @OneToOne(fetch = FetchType.LAZY)
    @JoinColumn(
            name = "venta_id",
            nullable = false,
            unique = true,
            foreignKey = @ForeignKey(name = "fk_viaje_venta")
    )
    private Venta venta;

    @OneToMany(
            mappedBy = "viaje",
            fetch = FetchType.LAZY
    )
    @Builder.Default
    private List<Tramo> tramos = new ArrayList<>();
}

package com.conforttravel.api.viaje.service.impl;

import com.conforttravel.api.shared.exception.ConflictException;
import com.conforttravel.api.shared.exception.ResourceNotFoundException;
import com.conforttravel.api.venta.entity.Venta;
import com.conforttravel.api.venta.repository.VentaRepository;
import com.conforttravel.api.viaje.dto.ViajeRequest;
import com.conforttravel.api.viaje.dto.ViajeResponse;
import com.conforttravel.api.viaje.entity.Viaje;
import com.conforttravel.api.viaje.mapper.ViajeMapper;
import com.conforttravel.api.viaje.repository.ViajeRepository;
import com.conforttravel.api.viaje.service.ViajeService;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true)
public class ViajeServiceImpl implements ViajeService {

    private final ViajeRepository viajeRepository;
    private final VentaRepository ventaRepository;
    private final ViajeMapper viajeMapper;

    @Override
    @Transactional
    public ViajeResponse crear(ViajeRequest request) {

        Venta venta = ventaRepository.findById(request.ventaId())
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "Venta no encontrada con id: "
                                        + request.ventaId()
                        )
                );

        if (viajeRepository.existsByVentaId(request.ventaId())) {
            throw new ConflictException(
                    "La venta con id "
                            + request.ventaId()
                            + " ya tiene un viaje asociado"
            );
        }

        Viaje viaje = viajeMapper.toEntity(request);
        viaje.setVenta(venta);

        Viaje guardado = viajeRepository.save(viaje);

        return viajeMapper.toResponseDto(guardado);
    }

    @Override
    public ViajeResponse obtenerPorId(Long id) {
        return viajeMapper.toResponseDto(obtenerEntidad(id));
    }

    @Override
    public List<ViajeResponse> obtenerTodos() {
        return viajeRepository.findAll()
                .stream()
                .map(viajeMapper::toResponseDto)
                .toList();
    }

    @Override
    @Transactional
    public void eliminar(Long id) {
        Viaje viaje = obtenerEntidad(id);

        viajeRepository.delete(viaje);
    }

    private Viaje obtenerEntidad(Long id) {
        return viajeRepository.findById(id)
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "Viaje no encontrado con id: " + id
                        )
                );
    }
}
