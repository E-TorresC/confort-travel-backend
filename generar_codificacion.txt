# Script: Generar árbol + contenido de archivos del proyecto (compatible PS 5.1)

$rootPath = "E:\PROYECTOS_FULLSTACK\confort_travel\confort-travel-backend"
$outputFile = Join-Path $rootPath "codigo_hasta_fase_18.txt"

$includePatterns = @("*.java", "*.xml", "*.yml", "*.yaml", "*.properties", "*.sql", "*.json")
$excludeDirs = @("target", ".git", ".idea", ".mvn", "node_modules", ".settings", "bin")

# --- Crear StreamWriter UNA sola vez ---
$utf8NoBom = New-Object System.Text.UTF8Encoding($false)
$stream = New-Object System.IO.StreamWriter($outputFile, $false, $utf8NoBom)
$stream.AutoFlush = $true

function Write-Line {
    param([string]$text = "")
    if ($null -eq $text) { $text = "" }
    $stream.WriteLine($text)
}

try {
    Write-Line "======================================================================"
    Write-Line "   PROYECTO: CONFORT TRAVEL BACKEND - CODIGO HASTA FASE 18"
    Write-Line "   Generado: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"
    Write-Line "   Ruta: $rootPath"
    Write-Line "======================================================================"
    Write-Line ""

    # ============ 1) ÁRBOL DE DIRECTORIOS ============
    Write-Line "======================================================================"
    Write-Line "   ESTRUCTURA DE DIRECTORIOS (ARBOL)"
    Write-Line "======================================================================"
    Write-Line ""

    function Show-Tree {
        param([string]$Path, [string]$Prefix = "")

        $items = Get-ChildItem -Path $Path -Force | Where-Object {
            $excludeDirs -notcontains $_.Name -and $_.Name -notlike "*.iml"
        } | Sort-Object @{Expression={$_.PSIsContainer};Descending=$true}, Name

        for ($i = 0; $i -lt $items.Count; $i++) {
            $item = $items[$i]
            $isLast = ($i -eq $items.Count - 1)
            $connector = if ($isLast) { "└── " } else { "├── " }
            $newPrefix = $Prefix + $(if ($isLast) { "    " } else { "│   " })

            Write-Line "$Prefix$connector$($item.Name)"

            if ($item.PSIsContainer) {
                Show-Tree -Path $item.FullName -Prefix $newPrefix
            }
        }
    }

    Write-Line (Split-Path $rootPath -Leaf)
    Show-Tree -Path $rootPath
    Write-Line ""

    # ============ 2) CONTENIDO DE ARCHIVOS ============
    Write-Line "======================================================================"
    Write-Line "   CONTENIDO DE ARCHIVOS"
    Write-Line "======================================================================"
    Write-Line ""

    $allFiles = Get-ChildItem -Path $rootPath -Recurse -File -Force | Where-Object {
        $file = $_

        $excluded = $false
        foreach ($dir in $excludeDirs) {
            if ($file.FullName -match [regex]::Escape("\$dir\")) {
                $excluded = $true
                break
            }
        }
        if ($excluded) { return $false }

        foreach ($pat in $includePatterns) {
            if ($file.Name -like $pat) { return $true }
        }
        return $false
    } | Sort-Object FullName

    $total = $allFiles.Count
    $counter = 0

    foreach ($file in $allFiles) {
        $counter++
        $relative = $file.FullName.Substring($rootPath.Length).TrimStart('\')

        Write-Line "======================================================================"
        Write-Line " ARCHIVO [$counter/$total]: $relative"
        Write-Line " Ruta completa: $($file.FullName)"
        Write-Line " Tamaño: $($file.Length) bytes"
        Write-Line "======================================================================"

        try {
            $content = Get-Content -Path $file.FullName -Raw -Encoding UTF8 -ErrorAction Stop
            if ([string]::IsNullOrEmpty($content)) {
                Write-Line "(ARCHIVO VACIO)"
            } else {
                Write-Line $content
            }
        }
        catch {
            Write-Line "(ERROR AL LEER: $($_.Exception.Message))"
        }

        Write-Line ""
        Write-Line ""
    }

    # ============ 3) RESUMEN FINAL ============
    Write-Line "======================================================================"
    Write-Line "   RESUMEN"
    Write-Line "======================================================================"
    Write-Line " Total de archivos procesados: $total"
    Write-Line " Archivo generado en: $outputFile"
    Write-Line " Fecha: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"
    Write-Line "======================================================================"
}
finally {
    $stream.Flush()
    $stream.Close()
    $stream.Dispose()
}

Write-Host ""
Write-Host "=============================================" -ForegroundColor Green
Write-Host " LISTO! Archivo generado:" -ForegroundColor Green
Write-Host " $outputFile" -ForegroundColor Yellow
Write-Host " Total archivos: $total" -ForegroundColor Cyan
Write-Host "=============================================" -ForegroundColor Green
Write-Host ""
