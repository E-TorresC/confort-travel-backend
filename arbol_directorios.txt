# Script: Generar SOLO el árbol de directorios del proyecto (compatible PS 5.1)

$rootPath = "E:\PROYECTOS_FULLSTACK\confort_travel\confort-travel-backend"
$outputFile = Join-Path $rootPath "estructura_proyecto.txt"

# Directorios a excluir del árbol
$excludeDirs = @("target", ".git", ".idea", ".mvn", "node_modules", ".settings", "bin", "out", "dist", "build")

# Extensiones/archivos a excluir (opcional: mostrar solo archivos relevantes)
# Si quieres mostrar TODOS los archivos, deja este array vacío: @()
$includePatterns = @("*.java", "*.xml", "*.yml", "*.yaml", "*.properties", "*.sql", "*.json", "*.md", "*.txt", "*.gradle", "*.iml")

# --- Crear StreamWriter UNA sola vez ---
$utf8NoBom = New-Object System.Text.UTF8Encoding($false)
$stream = New-Object System.IO.StreamWriter($outputFile, $false, $utf8NoBom)
$stream.AutoFlush = $true

function Write-Line {
    param([string]$text = "")
    if ($null -eq $text) { $text = "" }
    $stream.WriteLine($text)
}

try {
    Write-Line "======================================================================"
    Write-Line "   PROYECTO: CONFORT TRAVEL BACKEND"
    Write-Line "   ESTRUCTURA DE DIRECTORIOS (ARBOL)"
    Write-Line "   Generado: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"
    Write-Line "   Ruta: $rootPath"
    Write-Line "======================================================================"
    Write-Line ""

    # ============ ARBOL DE DIRECTORIOS ============
    function Show-Tree {
        param([string]$Path, [string]$Prefix = "")

        $items = Get-ChildItem -Path $Path -Force | Where-Object {
            # Excluir directorios no deseados
            if ($_.PSIsContainer -and ($excludeDirs -contains $_.Name)) { return $false }
            # Excluir archivos .iml
            if (-not $_.PSIsContainer -and $_.Name -like "*.iml") { return $false }
            # Excluir archivos que no coincidan con los patrones (solo si hay patrones definidos)
            if (-not $_.PSIsContainer -and $includePatterns.Count -gt 0) {
                foreach ($pat in $includePatterns) {
                    if ($_.Name -like $pat) { return $true }
                }
                return $false
            }
            return $true
        } | Sort-Object @{Expression={-not $_.PSIsContainer}}, Name

        for ($i = 0; $i -lt $items.Count; $i++) {
            $item = $items[$i]
            $isLast = ($i -eq $items.Count - 1)
            $connector = if ($isLast) { "└── " } else { "├── " }
            $newPrefix = $Prefix + $(if ($isLast) { "    " } else { "│   " })

            Write-Line "$Prefix$connector$($item.Name)"

            if ($item.PSIsContainer) {
                Show-Tree -Path $item.FullName -Prefix $newPrefix
            }
        }
    }

    # Raíz del proyecto
    Write-Line (Split-Path $rootPath -Leaf)
    Show-Tree -Path $rootPath
    Write-Line ""

    # ============ RESUMEN ============
    Write-Line "======================================================================"
    Write-Line "   RESUMEN"
    Write-Line "======================================================================"
    Write-Line " Archivo generado en: $outputFile"
    Write-Line " Fecha: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"
    Write-Line "======================================================================"
}
finally {
    $stream.Flush()
    $stream.Close()
    $stream.Dispose()
}

Write-Host ""
Write-Host "=============================================" -ForegroundColor Green
Write-Host " LISTO! Archivo generado:" -ForegroundColor Green
Write-Host " $outputFile" -ForegroundColor Yellow
Write-Host "=============================================" -ForegroundColor Green
Write-Host ""
