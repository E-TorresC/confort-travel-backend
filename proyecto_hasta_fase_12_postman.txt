package com.conforttravel.api.aerolinea.entity;

import jakarta.persistence.*;
import lombok.*;

import java.time.LocalDateTime;

@Entity
@Table(
        name = "aerolinea",
        uniqueConstraints = {
                @UniqueConstraint(
                        name = "uk_aerolinea_nombre",
                        columnNames = "nombre"
                ),
                @UniqueConstraint(
                        name = "uk_aerolinea_codigo",
                        columnNames = "codigo"
                )
        }
)
@Getter
@Setter
@NoArgsConstructor
@AllArgsConstructor
@Builder
public class Aerolinea {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(nullable = false, length = 100)
    private String nombre;

    @Column(length = 20)
    private String codigo;

    @Column(nullable = false)
    @Builder.Default
    private Boolean activo = true;

    @Column(name = "created_at", nullable = false)
    private LocalDateTime createdAt;

    @Column(name = "updated_at", nullable = false)
    private LocalDateTime updatedAt;

    @PrePersist
    protected void onCreate() {
        LocalDateTime now = LocalDateTime.now();
        createdAt = now;
        updatedAt = now;
    }

    @PreUpdate
    protected void onUpdate() {
        updatedAt = LocalDateTime.now();
    }
}

package com.conforttravel.api.aerolinea.service.impl;

import com.conforttravel.api.aerolinea.dto.AerolineaRequest;
import com.conforttravel.api.aerolinea.dto.AerolineaResponse;
import com.conforttravel.api.aerolinea.entity.Aerolinea;
import com.conforttravel.api.aerolinea.mapper.AerolineaMapper;
import com.conforttravel.api.aerolinea.repository.AerolineaRepository;
import com.conforttravel.api.aerolinea.service.AerolineaService;
import com.conforttravel.api.shared.exception.ResourceNotFoundException;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true)
public class AerolineaServiceImpl implements AerolineaService {

    private final AerolineaRepository aerolineaRepository;
    private final AerolineaMapper aerolineaMapper;

    @Override
    @Transactional
    public AerolineaResponse crear(AerolineaRequest request) {
        Aerolinea aerolinea = aerolineaMapper.toEntity(request);

        if (aerolinea.getActivo() == null) {
            aerolinea.setActivo(true);
        }

        Aerolinea guardada = aerolineaRepository.save(aerolinea);

        return aerolineaMapper.toResponseDto(guardada);
    }

    @Override
    public AerolineaResponse obtenerPorId(Long id) {
        return aerolineaMapper.toResponseDto(obtenerEntidad(id));
    }

    @Override
    public List<AerolineaResponse> obtenerTodos() {
        return aerolineaRepository.findAll()
                .stream()
                .map(aerolineaMapper::toResponseDto)
                .toList();
    }

    @Override
    @Transactional
    public AerolineaResponse actualizar(Long id, AerolineaRequest request) {
        Aerolinea aerolinea = obtenerEntidad(id);

        aerolinea.setNombre(request.nombre());
        aerolinea.setCodigo(request.codigo());

        if (request.activo() != null) {
            aerolinea.setActivo(request.activo());
        }

        Aerolinea actualizada = aerolineaRepository.save(aerolinea);

        return aerolineaMapper.toResponseDto(actualizada);
    }

    @Override
    @Transactional
    public void eliminar(Long id) {
        Aerolinea aerolinea = obtenerEntidad(id);

        aerolineaRepository.delete(aerolinea);
    }

    private Aerolinea obtenerEntidad(Long id) {
        return aerolineaRepository.findById(id)
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "AerolÃ­nea no encontrada con id: " + id
                        )
                );
    }
}

package com.conforttravel.api.checkin.service.impl;

import com.conforttravel.api.checkin.dto.CheckInRequest;
import com.conforttravel.api.checkin.dto.CheckInResponse;
import com.conforttravel.api.checkin.entity.CheckIn;
import com.conforttravel.api.checkin.entity.EstadoCheckIn;
import com.conforttravel.api.checkin.mapper.CheckInMapper;
import com.conforttravel.api.checkin.repository.CheckInRepository;
import com.conforttravel.api.checkin.service.CheckInService;
import com.conforttravel.api.shared.exception.BusinessException;
import com.conforttravel.api.shared.exception.ResourceNotFoundException;
import com.conforttravel.api.tramo.entity.Tramo;
import com.conforttravel.api.tramo.repository.TramoRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.OffsetDateTime;
import java.util.List;

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true)
public class CheckInServiceImpl implements CheckInService {

    private final CheckInRepository checkInRepository;
    private final TramoRepository tramoRepository;
    private final CheckInMapper checkInMapper;

    @Override
    @Transactional
    public CheckInResponse crear(CheckInRequest request) {
        Tramo tramo = tramoRepository.findById(request.tramoId())
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "Tramo no encontrado con id: " + request.tramoId()
                        )
                );

        CheckIn checkIn = checkInMapper.toEntity(request);

        checkIn.setTramo(tramo);
        checkIn.setEstado(EstadoCheckIn.PENDIENTE);

        /*
         * La fecha disponible serÃ¡ calculada por la lÃ³gica de negocio
         * correspondiente al tipo de viaje y aerolÃ­nea.
         *
         * Esta lÃ³gica serÃ¡ desarrollada en la fase de reglas de negocio.
         */
        checkIn.setFechaDisponible(tramo.getFechaHoraSalida());

        CheckIn guardado = checkInRepository.save(checkIn);

        return checkInMapper.toResponseDto(guardado);
    }

    @Override
    public CheckInResponse obtenerPorId(Long id) {
        return checkInMapper.toResponseDto(obtenerEntidad(id));
    }

    @Override
    public List<CheckInResponse> obtenerTodos() {
        return checkInRepository.findAll()
                .stream()
                .map(checkInMapper::toResponseDto)
                .toList();
    }

    @Override
    @Transactional
    public CheckInResponse notificar(Long id) {
        CheckIn checkIn = obtenerEntidad(id);

        if (checkIn.getEstado() != EstadoCheckIn.PENDIENTE) {
            throw new BusinessException(
                    "El check-in solo puede pasar de PENDIENTE a NOTIFICADO"
            );
        }

        checkIn.setEstado(EstadoCheckIn.NOTIFICADO);
        checkIn.setFechaNotificacion(OffsetDateTime.now());

        CheckIn actualizado = checkInRepository.save(checkIn);

        return checkInMapper.toResponseDto(actualizado);
    }

    @Override
    @Transactional
    public CheckInResponse atender(Long id) {
        CheckIn checkIn = obtenerEntidad(id);

        if (checkIn.getEstado() != EstadoCheckIn.NOTIFICADO) {
            throw new BusinessException(
                    "El check-in solo puede pasar de NOTIFICADO a ATENDIDO"
            );
        }

        checkIn.setEstado(EstadoCheckIn.ATENDIDO);
        checkIn.setFechaAtencion(OffsetDateTime.now());

        CheckIn actualizado = checkInRepository.save(checkIn);

        return checkInMapper.toResponseDto(actualizado);
    }

    private CheckIn obtenerEntidad(Long id) {
        return checkInRepository.findById(id)
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "Check-in no encontrado con id: " + id
                        )
                );
    }
}

package com.conforttravel.api.shared.exception;

import jakarta.servlet.http.HttpServletRequest;
import jakarta.validation.ConstraintViolationException;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

import java.time.OffsetDateTime;
import java.util.stream.Collectors;

@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(ResourceNotFoundException.class)
    public ResponseEntity<ApiErrorResponse> handleResourceNotFound(
            ResourceNotFoundException exception,
            HttpServletRequest request
    ) {
        return buildResponse(
                HttpStatus.NOT_FOUND,
                "RESOURCE_NOT_FOUND",
                exception.getMessage(),
                request.getRequestURI()
        );
    }

    @ExceptionHandler(BusinessException.class)
    public ResponseEntity<ApiErrorResponse> handleBusinessException(
            BusinessException exception,
            HttpServletRequest request
    ) {
        return buildResponse(
                HttpStatus.UNPROCESSABLE_ENTITY,
                "BUSINESS_RULE_ERROR",
                exception.getMessage(),
                request.getRequestURI()
        );
    }

    @ExceptionHandler(ConflictException.class)
    public ResponseEntity<ApiErrorResponse> handleConflict(
            ConflictException exception,
            HttpServletRequest request
    ) {
        return buildResponse(
                HttpStatus.CONFLICT,
                "CONFLICT",
                exception.getMessage(),
                request.getRequestURI()
        );
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ApiErrorResponse> handleValidation(
            MethodArgumentNotValidException exception,
            HttpServletRequest request
    ) {
        String message = exception.getBindingResult()
                .getFieldErrors()
                .stream()
                .map(error -> error.getField() + ": " + error.getDefaultMessage())
                .collect(Collectors.joining("; "));

        return buildResponse(
                HttpStatus.BAD_REQUEST,
                "VALIDATION_ERROR",
                message,
                request.getRequestURI()
        );
    }

    @ExceptionHandler(ConstraintViolationException.class)
    public ResponseEntity<ApiErrorResponse> handleConstraintViolation(
            ConstraintViolationException exception,
            HttpServletRequest request
    ) {
        String message = exception.getConstraintViolations()
                .stream()
                .map(violation ->
                        violation.getPropertyPath() + ": " + violation.getMessage()
                )
                .collect(Collectors.joining("; "));

        return buildResponse(
                HttpStatus.BAD_REQUEST,
                "VALIDATION_ERROR",
                message,
                request.getRequestURI()
        );
    }

    @ExceptionHandler(DataIntegrityViolationException.class)
    public ResponseEntity<ApiErrorResponse> handleDataIntegrityViolation(
            DataIntegrityViolationException exception,
            HttpServletRequest request
    ) {
        return buildResponse(
                HttpStatus.CONFLICT,
                "DATA_INTEGRITY_ERROR",
                "La operaciÃ³n no puede completarse porque viola una restricciÃ³n de datos",
                request.getRequestURI()
        );
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<ApiErrorResponse> handleUnexpectedException(
            Exception exception,
            HttpServletRequest request
    ) {
        return buildResponse(
                HttpStatus.INTERNAL_SERVER_ERROR,
                "INTERNAL_SERVER_ERROR",
                "Ha ocurrido un error interno en el servidor",
                request.getRequestURI()
        );
    }

    private ResponseEntity<ApiErrorResponse> buildResponse(
            HttpStatus status,
            String code,
            String message,
            String path
    ) {
        ApiErrorResponse response = new ApiErrorResponse(
                OffsetDateTime.now(),
                status.value(),
                code,
                message,
                path
        );

        return ResponseEntity
                .status(status)
                .body(response);
    }
}

package com.conforttravel.api.tramo.service.impl;

import com.conforttravel.api.shared.exception.BusinessException;
import com.conforttravel.api.shared.exception.ResourceNotFoundException;
import com.conforttravel.api.tramo.dto.TramoRequest;
import com.conforttravel.api.tramo.dto.TramoResponse;
import com.conforttravel.api.tramo.entity.Tramo;
import com.conforttravel.api.tramo.mapper.TramoMapper;
import com.conforttravel.api.tramo.repository.TramoRepository;
import com.conforttravel.api.tramo.service.TramoService;
import com.conforttravel.api.viaje.entity.Viaje;
import com.conforttravel.api.viaje.repository.ViajeRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true)
public class TramoServiceImpl implements TramoService {

    private final TramoRepository tramoRepository;
    private final ViajeRepository viajeRepository;
    private final TramoMapper tramoMapper;

    @Override
    @Transactional
    public TramoResponse crear(TramoRequest request) {
        validarFechas(request);

        Viaje viaje = obtenerViaje(request.viajeId());

        Tramo tramo = tramoMapper.toEntity(request);
        tramo.setViaje(viaje);

        Tramo guardado = tramoRepository.save(tramo);

        return tramoMapper.toResponseDto(guardado);
    }

    @Override
    public TramoResponse obtenerPorId(Long id) {
        return tramoMapper.toResponseDto(obtenerEntidad(id));
    }

    @Override
    public List<TramoResponse> obtenerTodos() {
        return tramoRepository.findAll()
                .stream()
                .map(tramoMapper::toResponseDto)
                .toList();
    }

    @Override
    @Transactional
    public TramoResponse actualizar(
            Long id,
            TramoRequest request
    ) {
        validarFechas(request);

        Tramo tramo = obtenerEntidad(id);
        Viaje viaje = obtenerViaje(request.viajeId());

        tramo.setTipoTramo(request.tipoTramo());
        tramo.setTipoViaje(request.tipoViaje());
        tramo.setOrigen(request.origen());
        tramo.setDestino(request.destino());
        tramo.setFechaHoraSalida(request.fechaHoraSalida());
        tramo.setFechaHoraLlegada(request.fechaHoraLlegada());
        tramo.setViaje(viaje);

        Tramo actualizado = tramoRepository.save(tramo);

        return tramoMapper.toResponseDto(actualizado);
    }

    @Override
    @Transactional
    public void eliminar(Long id) {
        Tramo tramo = obtenerEntidad(id);

        tramoRepository.delete(tramo);
    }

    private Tramo obtenerEntidad(Long id) {
        return tramoRepository.findById(id)
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "Tramo no encontrado con id: " + id
                        )
                );
    }

    private Viaje obtenerViaje(Long id) {
        return viajeRepository.findById(id)
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "Viaje no encontrado con id: " + id
                        )
                );
    }

    private void validarFechas(TramoRequest request) {
        if (request.fechaHoraLlegada() != null
                && !request.fechaHoraLlegada()
                .isAfter(request.fechaHoraSalida())) {

            throw new BusinessException(
                    "La fecha/hora de llegada debe ser posterior "
                            + "a la fecha/hora de salida"
            );
        }
    }
}

    public BigDecimal getGanancia() {
        if (precioInicial == null || precioFinal == null) {
            return null;
        }

        return precioFinal.subtract(precioInicial);
    }

    @PrePersist
    protected void onCreate() {
        LocalDateTime now = LocalDateTime.now();

        if (fechaCompra == null) {
            fechaCompra = now;
        }

        createdAt = now;
        updatedAt = now;
    }

    @PreUpdate
    protected void onUpdate() {
        updatedAt = LocalDateTime.now();
    }

import java.util.List;

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true)
public class VentaServiceImpl implements VentaService {

    private final VentaRepository ventaRepository;
    private final ClienteRepository clienteRepository;
    private final UsuarioRepository usuarioRepository;
    private final VentaMapper ventaMapper;

    @Override
    @Transactional
    public VentaResponse crear(VentaRequest request) {
        validarPrecios(request);

        Cliente cliente = obtenerCliente(request.clienteId());
        Usuario agente = obtenerAgente(request.agenteId());

        Venta venta = ventaMapper.toEntity(request);

        venta.setCliente(cliente);
        venta.setAgente(agente);

        Venta guardada = ventaRepository.save(venta);

        return ventaMapper.toResponseDto(guardada);
    }

    @Override
    public VentaResponse obtenerPorId(Long id) {
        return ventaMapper.toResponseDto(obtenerEntidad(id));
    }

    @Override
    public List<VentaResponse> obtenerTodos() {
        return ventaRepository.findAll()
                .stream()
                .map(ventaMapper::toResponseDto)
                .toList();
    }

    @Override
    @Transactional
    public VentaResponse actualizar(
            Long id,
            VentaRequest request
    ) {
        validarPrecios(request);

        Venta venta = obtenerEntidad(id);

        Cliente cliente = obtenerCliente(request.clienteId());
        Usuario agente = obtenerAgente(request.agenteId());

        venta.setFechaCompra(request.fechaCompra());
        venta.setPrecioInicial(request.precioInicial());
        venta.setPrecioFinal(request.precioFinal());
        venta.setObservaciones(request.observaciones());
        venta.setAdultos(request.adultos());
        venta.setInfantes(request.infantes());
        venta.setCliente(cliente);
        venta.setAgente(agente);

        Venta actualizada = ventaRepository.save(venta);

        return ventaMapper.toResponseDto(actualizada);
    }

    @Override
    @Transactional
    public void eliminar(Long id) {
        Venta venta = obtenerEntidad(id);

        ventaRepository.delete(venta);
    }

    private Venta obtenerEntidad(Long id) {
        return ventaRepository.findById(id)
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "Venta no encontrada con id: " + id
                        )
                );
    }

    private Cliente obtenerCliente(Long id) {
        return clienteRepository.findById(id)
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "Cliente no encontrado con id: " + id
                        )
                );
    }

    private Usuario obtenerAgente(Long id) {
        return usuarioRepository.findById(id)
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "Usuario agente no encontrado con id: " + id
                        )
                );
    }

    private void validarPrecios(VentaRequest request) {
        BigDecimal precioInicial = request.precioInicial();
        BigDecimal precioFinal = request.precioFinal();

        if (precioFinal.compareTo(precioInicial) < 0) {
            throw new BusinessException(
                    "El precio final no puede ser menor que el precio inicial"
            );
        }
    }

import com.conforttravel.api.venta.repository.VentaServicioRepository;

import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true)
public class VentaServicioServiceImpl
        implements VentaServicioService {

    private final VentaServicioRepository ventaServicioRepository;
    private final VentaRepository ventaRepository;
    private final ServicioAdicionalRepository servicioRepository;
    private final VentaServicioMapper ventaServicioMapper;

    @Override
    @Transactional
    public VentaServicioResponse agregar(
            VentaServicioRequest request
    ) {
        Venta venta = ventaRepository.findById(request.ventaId())
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "Venta no encontrada con id: " + request.ventaId()
                        )
                );

        ServicioAdicional servicio =
                servicioRepository.findById(request.servicioAdicionalId())
                        .orElseThrow(() ->
                                new ResourceNotFoundException(
                                        "Servicio adicional no encontrado con id: "
                                                + request.servicioAdicionalId()
                                )
                        );

        if (!Boolean.TRUE.equals(servicio.getActivo())) {
            throw new BusinessException(
                    "No se puede agregar un servicio adicional inactivo"
            );
        }

        VentaServicioId id = new VentaServicioId(
                venta.getId(),
                servicio.getId()
        );

        if (ventaServicioRepository.existsById(id)) {
            throw new BusinessException(
                    "El servicio adicional ya estÃ¡ asociado a la venta"
            );
        }

        VentaServicio ventaServicio =
                ventaServicioMapper.toEntity(request);

        ventaServicio.setId(id);
        ventaServicio.setVenta(venta);
        ventaServicio.setServicioAdicional(servicio);

        VentaServicio guardado =
                ventaServicioRepository.save(ventaServicio);

        return ventaServicioMapper.toResponseDto(guardado);
    }

    @Override
    public VentaServicioResponse obtenerPorId(
            Long ventaId,
            Long servicioAdicionalId
    ) {
        VentaServicioId id =
                new VentaServicioId(ventaId, servicioAdicionalId);

        VentaServicio ventaServicio =
                ventaServicioRepository.findById(id)
                        .orElseThrow(() ->
                                new ResourceNotFoundException(
                                        "Servicio de venta no encontrado"
                                )
                        );

        return ventaServicioMapper.toResponseDto(ventaServicio);
    }

    @Override
    public List<VentaServicioResponse> obtenerTodos() {
        return ventaServicioRepository.findAll()
                .stream()
                .map(ventaServicioMapper::toResponseDto)
                .toList();
    }

    @Override
    @Transactional
    public void eliminar(
            Long ventaId,
            Long servicioAdicionalId
    ) {
        VentaServicioId id =
                new VentaServicioId(
                        ventaId,
                        servicioAdicionalId
                );

        if (!ventaServicioRepository.existsById(id)) {
            throw new ResourceNotFoundException(
                    "Servicio de venta no encontrado"
            );
        }

        ventaServicioRepository.deleteById(id);
    }
}
