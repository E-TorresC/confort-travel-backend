                        filter.activo()
                )
        );

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.RequiredArgsConstructor;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.security.web.authentication.WebAuthenticationDetailsSource;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;

@Component
@RequiredArgsConstructor
public class JwtAuthenticationFilter extends OncePerRequestFilter {

    private final JwtService jwtService;
    private final CustomUserDetailsService userDetailsService;

    @Override
    protected void doFilterInternal(
            HttpServletRequest request,
            HttpServletResponse response,
            FilterChain filterChain
    ) throws ServletException, IOException {

        final String authorizationHeader =
                request.getHeader("Authorization");

        if (authorizationHeader == null
                || !authorizationHeader.startsWith("Bearer ")) {

            filterChain.doFilter(request, response);
            return;
        }

        final String jwt =
                authorizationHeader.substring(7);

        try {

            final String username =
                    jwtService.extractUsername(jwt);

            if (username != null
                    && SecurityContextHolder
                    .getContext()
                    .getAuthentication() == null) {

                UserDetails userDetails =
                        userDetailsService.loadUserByUsername(username);

                if (jwtService.isTokenValid(jwt, userDetails)) {

                    UsernamePasswordAuthenticationToken authentication =
                            new UsernamePasswordAuthenticationToken(
                                    userDetails,
                                    null,
                                    userDetails.getAuthorities()
                            );

                    authentication.setDetails(
                            new WebAuthenticationDetailsSource()
                                    .buildDetails(request)
                    );

                    SecurityContextHolder
                            .getContext()
                            .setAuthentication(authentication);
                }
            }

        } catch (Exception ignored) {
            /*
             * Un token inválido no debe detener el filtro.
             * Spring Security determinará posteriormente
             * que la solicitud no está autenticada.
             */
        }

        filterChain.doFilter(request, response);
    }
}

package com.conforttravel.api.auth.security;

import io.jsonwebtoken.Claims;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.stereotype.Service;

import javax.crypto.SecretKey;
import java.nio.charset.StandardCharsets;
import java.util.Date;
import java.util.HashMap;
import java.util.Map;
import java.util.function.Function;

@Service
public class JwtService {

    private final String secret;
    private final long expiration;

    public JwtService(
            @Value("${security.jwt.secret}") String secret,
            @Value("${security.jwt.expiration}") long expiration
    ) {
        this.secret = secret;
        this.expiration = expiration;
    }

    public String generateToken(UserDetails userDetails) {

        Map<String, Object> claims = new HashMap<>();

        claims.put(
                "roles",
                userDetails.getAuthorities()
                        .stream()
                        .map(authority -> authority.getAuthority())
                        .toList()
        );

        return Jwts.builder()
                .claims(claims)
                .subject(userDetails.getUsername())
                .issuedAt(new Date())
                .expiration(new Date(System.currentTimeMillis() + expiration))
                .signWith(getSigningKey())
                .compact();
    }

    public String extractUsername(String token) {
        return extractClaim(token, Claims::getSubject);
    }

    public Date extractExpiration(String token) {
        return extractClaim(token, Claims::getExpiration);
    }

    public <T> T extractClaim(
            String token,
            Function<Claims, T> claimsResolver
    ) {
        Claims claims = extractAllClaims(token);
        return claimsResolver.apply(claims);
    }

    public boolean isTokenValid(
            String token,
            UserDetails userDetails
    ) {
        final String username = extractUsername(token);

        return username.equals(userDetails.getUsername())
                && !isTokenExpired(token);
    }

    private boolean isTokenExpired(String token) {
        return extractExpiration(token).before(new Date());
    }

    private Claims extractAllClaims(String token) {

        return Jwts.parser()
                .verifyWith(getSigningKey())
                .build()
                .parseSignedClaims(token)
                .getPayload();
    }

    private SecretKey getSigningKey() {

        byte[] keyBytes = secret.getBytes(StandardCharsets.UTF_8);

        return Keys.hmacShaKeyFor(keyBytes);
    }
}

import java.time.ZoneOffset;

@Service
@RequiredArgsConstructor
@Slf4j
public class CheckInNotificationServiceImpl
        implements CheckInNotificationService {

    private final CheckInRepository checkInRepository;
    private final NotificacionRepository notificacionRepository;

    @Override
    @Transactional
    public void procesarCheckInsDisponibles() {

        OffsetDateTime ahora = OffsetDateTime.now(ZoneOffset.UTC);

        List<CheckIn> checkInsDisponibles =
                checkInRepository
                        .findByEstadoAndFechaDisponibleLessThanEqual(
                                EstadoCheckIn.PENDIENTE,
                                ahora
                        );

        if (checkInsDisponibles.isEmpty()) {
            log.debug(
                    "Scheduler de check-in: no existen check-ins pendientes disponibles"
            );
            return;
        }

        for (CheckIn checkIn : checkInsDisponibles) {

            if (checkIn.getEstado() != EstadoCheckIn.PENDIENTE) {
                continue;
            }

            crearNotificacion(checkIn);

            checkIn.setEstado(EstadoCheckIn.NOTIFICADO);
            checkIn.setFechaNotificacion(ahora);

            checkInRepository.save(checkIn);

            log.info(
                    "Check-in {} notificado automáticamente al agente responsable. Tramo: {}",
                    checkIn.getId(),
                    checkIn.getTramo().getId()
            );
        }
    }

    private void crearNotificacion(CheckIn checkIn) {

        // Navegamos por el modelo de dominio para obtener el agente que realizó la venta
        Usuario agenteResponsable = checkIn.getTramo().getVenta().getAgente();

        Notificacion notificacion = Notificacion.builder()
                .tipo(TipoNotificacion.CHECK_IN_DISPONIBLE)
                .mensaje(
                        "El check-in del tramo "
                                + checkIn.getTramo().getId()
                                + " ya se encuentra disponible."
                )
                .fechaCreacion(OffsetDateTime.now(ZoneOffset.UTC))
                .usuario(agenteResponsable)
                .checkIn(checkIn)
                .build();

        notificacionRepository.save(notificacion);
    }
}

    public CheckInResponse crear(CheckInRequest request) {

        Tramo tramo = tramoRepository.findById(request.tramoId())
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "Tramo no encontrado con id: "
                                        + request.tramoId()
                        )
                );

        if (checkInRepository.existsByTramoId(request.tramoId())) {

            throw new ConflictException(
                    "El tramo ya tiene un seguimiento de check-in"
            );
        }

        OffsetDateTime fechaDisponible =
                calcularFechaDisponible(tramo);

        CheckIn checkIn = CheckIn.builder()
                .tramo(tramo)
                .estado(EstadoCheckIn.PENDIENTE)
                .fechaDisponible(fechaDisponible)
                .build();

        CheckIn guardado = checkInRepository.save(checkIn);

        auditLogService.registrar(
                "CREAR",
                "CHECK_IN",
                guardado.getId(),
                "Se creó el seguimiento de check-in para el tramo: "
                        + guardado.getTramo().getId()
        );

        return checkInMapper.toResponseDto(guardado);
    }

    @Override

    public CheckInResponse obtenerPorId(Long id) {

    @Transactional(readOnly = true)

    public List<CheckInResponse> obtenerPorEstado(

    public PageResponse<CheckInResponse> buscar(

    ) {
        Page<CheckIn> page = (estado != null)

        return PageResponse.from(

        );

            throw new ConflictException(

                            "antes de marcarlo como ATENDIDO"

        checkIn.setFechaAtencion(

        auditLogService.registrar(

            );
        }

        return tramo.getFechaHoraSalida()
                .minusHours(48);

        );

        specification = specification.and(
                ClienteSpecification.telefonoContains(filter.telefono())
        );

        return specification;

        if (montoTotalVendido == null) montoTotalVendido = BigDecimal.ZERO;

        BigDecimal gananciaAcumulada = ventaRepository.sumGananciaAcumulada();
        if (gananciaAcumulada == null) gananciaAcumulada = BigDecimal.ZERO;

        long checkInsPendientes = checkInRepository.countByEstado(EstadoCheckIn.PENDIENTE);
        long checkInsNotificados = checkInRepository.countByEstado(EstadoCheckIn.NOTIFICADO);
        long checkInsAtendidos = checkInRepository.countByEstado(EstadoCheckIn.ATENDIDO);

        return new DashboardResumenResponse(
                cantidadVentas,
                montoTotalVendido,
                gananciaAcumulada,
                checkInsPendientes,
                checkInsNotificados,
                checkInsAtendidos
        );
    }

    public List<ProximoViajeResponse> obtenerProximosViajes() {

        return tramoRepository.findTop5ByFechaHoraSalidaGreaterThanEqualOrderByFechaHoraSalidaAsc(

        ).stream().map(tramo -> new ProximoViajeResponse(

            Object[] fila

        Long id = ((Number) fila[0]).longValue();

        String nombre = (String) fila[1];

        Long cantidadVentas =
                ((Number) fila[2]).longValue();

        BigDecimal montoTotal =
                convertirBigDecimal(fila[3]);

        BigDecimal gananciaTotal =
                convertirBigDecimal(fila[4]);

        return new ReporteAgrupadoResponse(
                id,
                nombre,
                cantidadVentas,
                montoTotal,
                gananciaTotal
        );
    }


    private BigDecimal convertirBigDecimal(Object valor) {

        if (valor == null) {
            return BigDecimal.ZERO;
        }

        if (valor instanceof BigDecimal bigDecimal) {
            return bigDecimal;
        }

        return new BigDecimal(valor.toString());
    }


    private void validarRangoFechas(
            LocalDate fechaDesde,
            LocalDate fechaHasta
    ) {

        if (fechaDesde == null || fechaHasta == null) {
            throw new IllegalArgumentException(
                    "Las fechas desde y hasta son obligatorias"
            );
        }

        if (fechaDesde.isAfter(fechaHasta)) {
            throw new IllegalArgumentException(
                    "La fecha desde no puede ser posterior a la fecha hasta"
            );
        }
    }

    private OffsetDateTime inicioDelDia(LocalDate fecha) {
        return fecha.atStartOfDay(ZoneOffset.UTC).toOffsetDateTime();
    }

    private OffsetDateTime inicioDelDiaSiguiente(LocalDate fecha) {
        return fecha.plusDays(1).atStartOfDay(ZoneOffset.UTC).toOffsetDateTime();
    }

            Long id,
            ServicioAdicionalRequest request
    ) {

        servicio.setDescripcion(request.descripcion());

    @Override
    @Transactional
    public TramoResponse crear(TramoRequest request) {
        validarFechas(request);

        Venta venta = obtenerVenta(request.ventaId());

        Tramo tramo = tramoMapper.toEntity(request);
        tramo.setVenta(venta);

        Tramo guardado = tramoRepository.save(tramo);

        auditLogService.registrar(
                "CREAR",
                "TRAMO",
                guardado.getId(),
                "Se creó el tramo desde " + guardado.getOrigen() + " hacia " + guardado.getDestino()
        );

        checkInService.crear(new CheckInRequest(guardado.getId()));
        return tramoMapper.toResponseDto(guardado);
    }

    @Override
    public TramoResponse obtenerPorId(Long id) {
        return tramoMapper.toResponseDto(obtenerEntidad(id));
    }

    @Override
    public List<TramoResponse> obtenerTodos() {
        return tramoRepository.findAll().stream().map(tramoMapper::toResponseDto).toList();
    }

    @Override
    public List<TramoResponse> obtenerPorVenta(Long ventaId) {
        return tramoRepository.findByVentaId(ventaId).stream().map(tramoMapper::toResponseDto).toList();
    }

    @Override
    @Transactional
    public TramoResponse actualizar(Long id, TramoRequest request) {
        validarFechas(request);

        Tramo tramo = obtenerEntidad(id);
        Venta venta = obtenerVenta(request.ventaId());

        tramo.setTipoTramo(request.tipoTramo());
        tramo.setTipoViaje(request.tipoViaje());
        tramo.setOrigen(request.origen());
        tramo.setDestino(request.destino());
        tramo.setFechaHoraSalida(request.fechaHoraSalida());
        tramo.setFechaHoraLlegada(request.fechaHoraLlegada());
        tramo.setVenta(venta);

        Tramo actualizado = tramoRepository.save(tramo);

        auditLogService.registrar(
                "ACTUALIZAR",
                "TRAMO",
                actualizado.getId(),
                "Se actualizó el tramo desde " + actualizado.getOrigen() + " hacia " + actualizado.getDestino()
        );

        return tramoMapper.toResponseDto(actualizado);
    }

    @Override
    @Transactional
    public void eliminar(Long id) {
        Tramo tramo = obtenerEntidad(id);

        auditLogService.registrar(
                "ELIMINAR",
                "TRAMO",
                tramo.getId(),
                "Se eliminó el tramo desde " + tramo.getOrigen() + " hacia " + tramo.getDestino()
        );

        tramoRepository.delete(tramo);
    }

    private Tramo obtenerEntidad(Long id) {
        return tramoRepository.findById(id).orElseThrow(() -> new ResourceNotFoundException("Tramo no encontrado con id: " + id));
    }

    private Venta obtenerVenta(Long id) {
        return ventaRepository.findById(id).orElseThrow(() -> new ResourceNotFoundException("Venta no encontrada con id: " + id));
    }

    private void validarFechas(TramoRequest request) {
        if (request.fechaHoraLlegada() != null && !request.fechaHoraLlegada().isAfter(request.fechaHoraSalida())) {
            throw new BusinessException("La fecha/hora de llegada debe ser posterior a la fecha/hora de salida");
        }
    }

        );

                UsuarioSpecification.activoEquals(filter.activo())
        );

        return specification;
    }

                        new ResourceNotFoundException(
                                "Venta no encontrada con id: " + request.ventaId()
                        )

                                )
                        );

            );

                servicio.getId()

            );

                ventaServicioMapper.toEntity(request);

        ventaServicio.setServicioAdicional(servicio);

        VentaServicio guardado =
                ventaServicioRepository.save(ventaServicio);

        auditLogService.registrar(
                "AGREGAR",
                "VENTA_SERVICIO",
                venta.getId(),
                "Se agregó el servicio adicional "

@Component
public class VentaSpecificationBuilder {

    public Specification<Venta> build(
            VentaFilterRequest filter
    ) {

        Specification<Venta> specification =
                Specification.allOf();

        if (filter == null) {
            return specification;
        }

        specification = specification.and(
                VentaSpecification.fechaCompraDesde(
                        filter.fechaDesde()
                )
        );

        specification = specification.and(
                VentaSpecification.fechaCompraHasta(
                        filter.fechaHasta()
                )
        );

        specification = specification.and(
                VentaSpecification.precioFinalMinimo(
                        filter.precioMinimo()
                )
        );

                VentaSpecification.precioFinalMaximo(
                        filter.precioMaximo()
                )
        );

        specification = specification.and(
                VentaSpecification.clienteIdEquals(
                        filter.clienteId()
                )
        );

        specification = specification.and(
                VentaSpecification.agenteIdEquals(
                        filter.agenteId()
                )
        );

        return specification;
    }
